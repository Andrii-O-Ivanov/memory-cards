import { createSlice } from '@reduxjs/toolkit';

/**
 * Завантажує історію рекордів із локального сховища браузера (localStorage).
 * Якщо дані відсутні або виникає помилка парсингу, повертає порожній масив.
 *
 * @function
 * @returns {Array<Object>} Масив збережених рекордів або порожній масив.
 */
const loadLeaderboard = () => {
    try {
        const saved = localStorage.getItem('memory-game-leaderboard');
        return saved ? JSON.parse(saved) : [];
    } catch (e) {
        console.error("Помилка завантаження рекордів:", e);
        return [];
    }
};

/**
 * @typedef {Object} GameState
 * @property {string} username - Ім'я поточного гравця.
 * @property {string} difficulty - Обрана складність гри (кількість карток).
 * @property {('idle'|'playing'|'finished')} gameStatus - Поточний статус ігрового циклу.
 * @property {Object} score - Поточний рахунок гравця.
 * @property {number} score.time - Витрачений час у секундах.
 * @property {number} score.moves - Кількість зроблених ходів.
 * @property {Array<Object>} leaderboard - Масив найкращих результатів (Топ-10).
 */
const initialState = {
  username: '',
  difficulty: '12',
  gameStatus: 'idle',
  score: {
    time: 0,
    moves: 0
  },
  leaderboard: loadLeaderboard()
};

/**
 * Redux-слайс для управління глобальним станом гри "Memory Cards".
 * Включає логіку налаштування гри, збереження рахунку та управління таблицею лідерів.
 */
const gameSlice = createSlice({
  name: 'game',
  initialState,
  reducers: {
    /**
     * Встановлює налаштування перед початком гри та змінює статус на 'playing'.
     * @param {GameState} state - Поточний стан гри.
     * @param {Object} action - Action об'єкт.
     * @param {Object} action.payload - Дані налаштувань.
     * @param {string} action.payload.username - Ім'я гравця.
     * @param {string} action.payload.difficulty - Складність гри.
     */
    setGameSettings: (state, action) => {
      state.username = action.payload.username;
      state.difficulty = action.payload.difficulty;
      state.gameStatus = 'playing';
    },
    
    /**
     * Завершує гру, зберігає фінальний рахунок, додає його в таблицю лідерів
     * з урахуванням сортування (Складність -> Час -> Ходи) та зберігає в localStorage.
     * @param {GameState} state - Поточний стан гри.
     * @param {Object} action - Action об'єкт.
     * @param {Object} action.payload - Фінальний результат.
     * @param {number} action.payload.time - Фінальний час.
     * @param {number} action.payload.moves - Фінальна кількість ходів.
     */
    finishGame: (state, action) => {
      state.gameStatus = 'finished';
      state.score = action.payload;

      const newRecord = {
        id: Date.now(),
        username: state.username,
        difficulty: state.difficulty,
        time: action.payload.time,
        moves: action.payload.moves,
        date: new Date().toLocaleDateString()
      };

      state.leaderboard.push(newRecord);

      // Сортування: Складність (за спаданням) -> Час (за зростанням) -> Ходи
      state.leaderboard.sort((a, b) => {
          const diffA = Number(a.difficulty);
          const diffB = Number(b.difficulty);
          
          if (diffA !== diffB) return diffB - diffA;
          if (a.time !== b.time) return a.time - b.time;
          return a.moves - b.moves;
      });

      // Зберігаємо лише топ-10 результатів
      if (state.leaderboard.length > 10) {
          state.leaderboard = state.leaderboard.slice(0, 10);
      }

      localStorage.setItem('memory-game-leaderboard', JSON.stringify(state.leaderboard));
    },

    /**
     * Скидає стан гри до початкового (idle) та обнуляє лічильники часу і ходів.
     * @param {GameState} state - Поточний стан гри.
     */
    resetGame: (state) => {
      state.gameStatus = 'idle';
      state.score = { time: 0, moves: 0 };
    },

    /**
     * Повністю очищує таблицю лідерів у стані Redux та видаляє дані з localStorage.
     * @param {GameState} state - Поточний стан гри.
     */
    clearLeaderboard: (state) => {
      state.leaderboard = [];
      localStorage.removeItem('memory-game-leaderboard');
    }
  }
});

export const { setGameSettings, finishGame, resetGame, clearLeaderboard } = gameSlice.actions;
export default gameSlice.reducer;
